CREPE Integration — SING Mode

Replacing FFT autocorrelation with TensorFlow.js CREPE for ML-based monophonic pitch detection  ·  Target file: music-theory-pro-MASTER.html  ·  May 2026
Why CREPE

The current SING mode uses FFT autocorrelation (_singLoop) — accurate for clean guitar but struggles with: singing voice harmonics, noisy environments, guitar with vibrato/bends, and ambiguous partials. CREPE (Convolutional Representation for Pitch Estimation) is a pretrained deep learning model trained on 11,000+ hours of audio. It outputs a per-frame pitch estimate with confidence score — dramatically outperforming autocorrelation for both voice and guitar.

Scope: CREPE replaces only the monophonic pitch detection pathway inside _singLoop. The HPS + chroma polyphonic chord detector, KCM.bus broadcasts, WAV capture, and all display logic remain untouched.

Phase 1 — CDN Load Code

Add to <head>, after existing scripts:

/* CREPE — ML pitch detection */
/* Load order: TF core → TF WebGL backend → CREPE model */
<script src="https://cdn.jsdelivr.net/npm/@tensorflow/tfjs@4.17.0/dist/tf.min.js"></script>
<script src="https://cdn.jsdelivr.net/npm/@tensorflow-models/pitch-detection@0.0.2/dist/pitch-detection.min.js"></script>

/* In JS — init after AudioContext is created */
let _crepeModel = null;
let _crepeReady = false;

async function _loadCREPE() {
  try {
    // pitchDetection is the @tensorflow-models/pitch-detection API
    _crepeModel = await pitchDetection.load(
      pitchDetection.SupportedModels.CREPE,
      { audioContext: _audioCtx, sampleRate: _audioCtx.sampleRate }
    );
    _crepeReady = true;
    console.log('[KCM] CREPE model ready');
    // Update UI — replace loading spinner in SING panel
    const badge = document.getElementById('singCrepeStatus');
    if (badge) badge.textContent = 'ML PITCH ✓';
  } catch (e) {
    console.warn('[KCM] CREPE load failed, falling back to FFT', e);
    _crepeReady = false; // graceful fallback
  }
}

/* Call after AudioContext unlock (inside startAudioEngine) */
_loadCREPE();
Phase 2 — _singLoop Swap

The current autocorrelation block occupies roughly 40 lines inside _singLoop. Replace only that block — the confidence score gates the silence check, everything else flows identically:

/* === BEFORE (FFT autocorrelation — remove this block) ===
   const buf = new Float32Array(_SING_FFT);
   _singAnalyser.getFloatTimeDomainData(buf);
   const rms = Math.sqrt(buf.reduce((s,v)=>s+v*v,0)/buf.length);
   if (rms < _SING_SILENCE_GATE) { ... silence handling ... return; }
   const freq = _autocorrelate(buf, _audioCtx.sampleRate);
   ... === END REMOVE === */

/* === AFTER (CREPE inference — insert this block) === */
async function _singLoop() {
  if (!_singActive) return;

  if (_crepeReady && _crepeModel) {
    // CREPE path — async per-frame inference
    const pitch = await _crepeModel.estimatePitch();
    // pitch = { frequency: 440.2, confidence: 0.94 } | null

    if (!pitch || pitch.confidence < _SING_CONFIDENCE_MIN) {
      _handleSilence(); // existing silence handler
    } else {
      _handlePitch(pitch.frequency); // existing pitch handler — unchanged
    }
  } else {
    // Fallback — existing FFT autocorrelation (unchanged)
    _singLoopFFT();
  }

  _singRAF = requestAnimationFrame(_singLoop);
}

_SING_CONFIDENCE_MIN = 0.70 — CREPE returns 0–1. At 0.70, the model is quite sure. Tune down to 0.60 if too many valid notes get gated; up to 0.80 in noisy rooms. This replaces the amplitude RMS silence gate for the CREPE path only.

Phase 3 — Guitar / Voice Constant Sets
const SING_PRESETS = {
  guitar: {
    lockFrames:  3,
    holdGap:     250,   // ms
    minDur:      120,   // ms
    cooldown:    40,    // ms
    smoothDecay: 0.60,
    smoothRise:  0.40,
    confidenceMin: 0.72,
  },
  voice: {
    lockFrames:  5,    // voice transitions slower
    holdGap:     350,   // longer inter-note gap
    minDur:      180,   // longer minimum note
    cooldown:    60,
    smoothDecay: 0.72,  // voice vibrato — slower decay
    smoothRise:  0.28,
    confidenceMin: 0.65, // voice more ambiguous — slightly lower gate
  }
};

let _singPreset = 'guitar'; // current default

/* Toggle button in SING panel — updates _singPreset and live constants */
function setSingMode(mode) {
  _singPreset = mode;
  const p = SING_PRESETS[mode];
  _SING_LOCK_FRAMES   = p.lockFrames;
  _SING_HOLD_GAP      = p.holdGap;
  _SING_MIN_DUR       = p.minDur;
  _SING_COOLDOWN      = p.cooldown;
  _SING_SMOOTH_DECAY  = p.smoothDecay;
  _SING_SMOOTH_RISE   = p.smoothRise;
  _SING_CONFIDENCE_MIN = p.confidenceMin;
}
